Seamless AI Agent Integration

Drop Gosh into your current agent pipelines with zero code rewrites. 

# Set Gosh as your primary agent tool execution shell
export SHELL=/usr/local/bin/gosh

# Optional: Enable session continuity across tool invocations
export GOSH_SESSION_ID="agent-session-$(date +%s)"
import subprocess, os

env = os.environ.copy()
env["GOSH_SESSION_ID"] = "autonomous-agent-42"

def agent_exec(cmd: str) -> str:
    res = subprocess.run(["gosh", "-c", cmd], capture_output=True, text=True, env=env)
    if res.returncode != 0: raise RuntimeError(res.stderr)
    return res.stdout.strip()

# State naturally persists across calls!
agent_exec("mkdir -p /tmp/demo_agent && cd /tmp/demo_agent")
agent_exec("git init -q && git config user.name 'GhostAgent'")
agent_exec("echo 'Autonomous execution verified' > status.txt && git add . && git commit -qm 'init'")
print(agent_exec("git log --oneline"))
